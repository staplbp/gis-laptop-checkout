// GIS Laptop Checkout — Apps Script Backend + Email Handler // Gmail sender: Meckgischeckout@gmail.com // Sheet ID: 1k5ZJFR7B-GTjJt9KH6S7PuWVknF5Wk05ieXusmUwNec const SHEET_ID = '1k5ZJFR7B-GTjJt9KH6S7PuWVknF5Wk05ieXusmUwNec'; const SHEET_NAME = 'Sheet1'; const SENDER = 'Meckgischeckout@gmail.com'; const SENDER_NAME = 'GIS Laptop Checkout'; const MGR_1 = 'Benita.Staples@mecklenburgcountync.gov'; const MGR_2 = 'Angela.Johnson@mecklenburgcountync.gov'; const HEADERS = [ 'id','name','email','checkoutStart','checkoutEnd','businessDays', 'purpose','status','submittedOn','laptop','approvedBy','approvedByEmail', 'approvedOn','returnedOn','note','reminderSent','overdueSent','requestedManager' ]; function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { const p = e.parameter || {}; let result; try { switch(p.action) { case 'getAll': result = getAllRequests(); break; case 'getByEmail': result = getByEmail(p.email); break; case 'add': result = addRequest(p); break; case 'update': result = updateRequest(p); break; case 'checkReminders': result = checkReminders(); break; default: result = { error: 'Unknown action: ' + p.action }; } } catch(err) { result = { error: err.toString() }; } return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // ── SHEET HELPERS ────────────────────────── function getSheet() { return SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); } function rowToObj(row) { const obj = {}; const dateFields = ['checkoutStart','checkoutEnd','submittedOn','approvedOn','returnedOn']; HEADERS.forEach((h, i) => { if (dateFields.includes(h) && row[i] instanceof Date) { const d = row[i]; const y = d.getFullYear(); const m = String(d.getMonth()+1).padStart(2,'0'); const day = String(d.getDate()).padStart(2,'0'); obj[h] = y + '-' + m + '-' + day; } else { obj[h] = row[i] !== undefined ? String(row[i]) : ''; } }); return obj; } function getAllRequests() { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; return data.slice(1).map(row => rowToObj(row)); } function getByEmail(email) { return getAllRequests().filter( r => r.email.toLowerCase() === (email || '').toLowerCase() ); } function addRequest(p) { const sheet = getSheet(); const row = HEADERS.map(h => p[h] || ''); sheet.appendRow(row); sendEmailNewRequest(p); return { success: true }; } function updateRequest(p) { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const idCol = HEADERS.indexOf('id'); for (let i = 1; i < data.length; i++) { if (String(data[i][idCol]) === String(p.id)) { HEADERS.forEach((h, j) => { if (p[h] !== undefined) sheet.getRange(i + 1, j + 1).setValue(p[h]); }); // Send decision email when status changes to approved or denied if (p.status === 'approved' || p.status === 'denied') { const updated = rowToObj(sheet.getRange(i + 1, 1, 1, HEADERS.length).getValues()[0]); sendEmailDecision(updated); } return { success: true }; } } return { error: 'Record not found' }; } // ── REMINDERS + OVERDUE ──────────────────── function checkReminders() { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const idCol = HEADERS.indexOf('id'); const today = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'); let notified = 0; for (let i = 1; i < data.length; i++) { const row = rowToObj(data[i]); if (row.status !== 'approved') continue; // Morning-of reminder if (row.checkoutEnd === today && row.reminderSent !== 'true') { sendEmailReminder(row); sheet.getRange(i + 1, HEADERS.indexOf('reminderSent') + 1).setValue('true'); notified++; } // Overdue notice if (row.checkoutEnd < today && row.overdueSent !== 'true' && row.returnedOn === '') { sendEmailOverdue(row); sheet.getRange(i + 1, HEADERS.indexOf('overdueSent') + 1).setValue('true'); notified++; } } return { success: true, notified: notified }; } // Trigger this daily at 7am from Apps Script triggers function dailyReminderCheck() { checkReminders(); } // ── EMAIL HELPERS ────────────────────────── function fmt(dateStr) { if (!dateStr) return ''; try { const parts = dateStr.split('-'); const d = new Date(parts[0], parts[1]-1, parts[2]); return Utilities.formatDate(d, Session.getScriptTimeZone(), 'MMM d, yyyy'); } catch(e) { return dateStr; } } function sendMail(to, subject, body) { GmailApp.sendEmail(to, subject, body, { name: SENDER_NAME, from: SENDER }); } function sendEmailNewRequest(r) { const subject = 'New Laptop Checkout Request from ' + r.name; const body = 'A new laptop checkout request has been submitted.\n\n' + 'Staff Name: ' + r.name + '\n' + 'Email: ' + r.email + '\n' + 'Dates: ' + fmt(r.checkoutStart) + ' to ' + fmt(r.checkoutEnd) + '\n' + 'Business Days: ' + r.businessDays + '\n' + 'Purpose: ' + r.purpose + '\n' + 'Submitted On: ' + fmt(r.submittedOn) + '\n\n' + 'Log in to the GIS Laptop Checkout app to approve or deny this request.'; sendMail(MGR_1 + ',' + MGR_2, subject, body); } function sendEmailDecision(r) { const approved = r.status === 'approved'; const subject = 'Your Laptop Request Has Been ' + (approved ? 'Approved' : 'Denied'); const body = 'Hi ' + r.name + ',\n\n' + 'Your laptop checkout request has been ' + r.status + '.\n\n' + (approved ? 'Laptop Assigned: Laptop ' + r.laptop + '\n' : '') + 'Checkout Dates: ' + fmt(r.checkoutStart) + ' to ' + fmt(r.checkoutEnd) + '\n' + 'Reviewed By: ' + r.approvedBy + '\n' + (r.note ? 'Note: ' + r.note + '\n' : '') + '\nIf you have questions, contact your manager directly.'; sendMail(r.email, subject, body); } function sendEmailReminder(r) { const subject = 'Reminder: Laptop Due Back Today'; const body = 'Hi ' + r.name + ',\n\n' + 'This is a reminder that Laptop ' + r.laptop + ' is due back today, ' + fmt(r.checkoutEnd) + '.\n\n' + 'Please return it to the GIS office today.\n' + 'If you need assistance, contact your manager directly.'; sendMail(r.email, subject, body); } function sendEmailOverdue(r) { const daysOverdue = Math.round( (new Date() - new Date(r.checkoutEnd)) / 86400000 ); const subject = 'OVERDUE: Laptop ' + r.laptop + ' Has Not Been Returned'; const body = 'Hi ' + r.name + ',\n\n' + 'Laptop ' + r.laptop + ' is now ' + daysOverdue + ' day(s) overdue.\n' + 'It was due back on ' + fmt(r.checkoutEnd) + '.\n\n' + 'Please return it to the GIS office immediately.\n' + 'Your manager ' + r.approvedBy + ' has also been notified.'; sendMail(r.email, subject, body); if (r.approvedByEmail) sendMail(r.approvedByEmail, subject, body); }
Setup Steps
1
Sign in to the new Gmail
Open a browser tab and sign in to Meckgischeckout@gmail.com. Keep this tab open.
2
Open your Google Sheet
While signed in as Meckgischeckout@gmail.com, go to the Google Sheet at docs.google.com/spreadsheets/d/1k5ZJFR7B-GTjJt9KH6S7PuWVknF5Wk05ieXusmUwNec. If it says you need access, share the sheet with Meckgischeckout@gmail.com as an Editor from your personal Gmail first.
3
Open Apps Script from the Sheet
Click Extensions then Apps Script. Delete all existing code. Click the Copy Script button above, then paste it in. Click Save (Ctrl+S).
4
Deploy as Web App
Click Deploy then New deployment. Select type Web app. Set Execute as to Me (Meckgischeckout@gmail.com). Set Who has access to Anyone. Click Deploy and authorize when prompted.
5
Paste the Web App URL here
Copy the Web App URL from the deployment screen. Paste it into the field above and click Save URL. Then click Test Connection.
6
Set up the daily reminder trigger
In Apps Script, click the clock icon (Triggers) in the left sidebar. Click Add Trigger. Set function to dailyReminderCheck. Set event source to Time-driven. Set type to Day timer. Set time to 7am to 8am. Click Save. This runs every morning automatically so reminders and overdue notices go out without anyone opening the app.
7
Share the Sheet with Angela
In your Google Sheet click Share. Add Angela's email and set her to Viewer.
8
Upload the updated index.html to GitHub
Replace the current file in your gis-laptop-checkout repo with this new version. Your live site updates within 2 minutes.
